name: Build ẞetaMusic Android APK from source ZIP

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: read

jobs:
  build-apk:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - name: Checkout tiny mobile-upload repository
        uses: actions/checkout@v4

      - name: Unpack Android source
        run: unzip -oq BetaMusic-Android-source.zip -d .

      - name: Configure Java
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Configure Android SDK
        uses: android-actions/setup-android@v3

      - name: Install Android SDK components
        run: sdkmanager "platforms;android-36" "build-tools;36.0.0" "platform-tools"

      - name: Configure Gradle 8.13
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.13'

      - name: Compile and test
        run: gradle testDebugUnitTest assembleDebug --stacktrace

      - name: Publish APK
        uses: actions/upload-artifact@v4
        with:
          name: betamusic-android-debug-apk
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
          retention-days: 14
